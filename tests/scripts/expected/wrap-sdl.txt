SDK alone: the DRM provider, and no SDL2
build-linux: dry run
  lane      x86_64-linux-gnu x86_64-linux-gnu-g++ linux-x86_64 none
  configure ./configure --target x86_64-linux-gnu --target-host --compiler x86_64-linux-gnu-g++ --sdk linux-x86_64 --runner native --build debug
  build     ./build --target apps/display-demo/
  image     out-target-x86_64-linux-gnu/apps/display-demo/display-demo
  provider platform.linux.defaults 1
  provider platform.linux.display 1
  provider platform.linux.fs 0
  provider platform.linux.imu 0
  provider platform.linux.mcu 1
  provider platform.linux.rtc 0
  provider platform.linux.stdio 0
  provider platform.linux.touch 0
  no-library SDL2
