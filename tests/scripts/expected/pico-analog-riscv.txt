build-pico: dry run
  lane      riscv32-pico-elf riscv32-pico-elf-gcc pico-riscv pico2-riscv
  configure ./configure --target riscv32-pico-elf --compiler riscv32-pico-elf-gcc --sdk pico-riscv --board pico2-riscv --build debug
  build     ./build --target apps/analog-smoke/
  image     out-target-riscv32-pico-elf/apps/analog-smoke/analog-smoke

  provider platform.pico.stdio 1
  abi mm_pico_mcu_adc_read

  uf2 rp2350

  control provider platform.pico.stdio 0
exit 0
